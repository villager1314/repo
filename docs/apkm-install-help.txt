usage: apkm install [-h] [--keep-apk] [--download-only]
                    [--output-dir OUTPUT_DIR]
                    names [names ...]

positional arguments:
  names                 软件名

options:
  -h, --help            show this help message and exit
  --keep-apk            安装成功后保留下载的 APK；默认核对安装版本后删除
  --download-only       只下载并校验
  --output-dir OUTPUT_DIR
                        下载目录
